namespace = paradox_agent_testbed

# Marks the human-controlled country as the bridge target.
country_event = {
	id = paradox_agent_testbed.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
	}

	immediate = {
		set_country_flag = paradox_agent_bridge_active
		log = "[PARADOX_AGENT]|BRIDGE_READY|date=[GetDate]|country=[This.GetName]"
	}
}

# Produces a deliberately simple line protocol for the external controller.
country_event = {
	id = paradox_agent_testbed.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = paradox_agent_bridge_active
		is_ai = no
	}

	immediate = {
		export_resource_stockpile_to_variable = { resource = energy variable = paradox_agent_energy }
		export_resource_stockpile_to_variable = { resource = minerals variable = paradox_agent_minerals }
		export_resource_stockpile_to_variable = { resource = food variable = paradox_agent_food }
		export_resource_stockpile_to_variable = { resource = consumer_goods variable = paradox_agent_consumer_goods }
		export_resource_stockpile_to_variable = { resource = alloys variable = paradox_agent_alloys }
		export_resource_stockpile_to_variable = { resource = unity variable = paradox_agent_unity }
		export_resource_stockpile_to_variable = { resource = influence variable = paradox_agent_influence }

		export_resource_income_to_variable = { resource = energy variable = paradox_agent_energy_income }
		export_resource_income_to_variable = { resource = minerals variable = paradox_agent_minerals_income }
		export_resource_income_to_variable = { resource = food variable = paradox_agent_food_income }
		export_resource_income_to_variable = { resource = consumer_goods variable = paradox_agent_consumer_goods_income }
		export_resource_income_to_variable = { resource = alloys variable = paradox_agent_alloys_income }
		export_resource_income_to_variable = { resource = unity variable = paradox_agent_unity_income }

		export_trigger_value_to_variable = {
			trigger = num_owned_planets
			variable = paradox_agent_planet_count
		}

		log = "[PARADOX_AGENT]|STATE_BEGIN|date=[GetDate]"
		log = "[PARADOX_AGENT]|COUNTRY|name=[This.GetName]|planets=[This.paradox_agent_planet_count]|energy=[This.paradox_agent_energy]|energy_income=[This.paradox_agent_energy_income]|minerals=[This.paradox_agent_minerals]|minerals_income=[This.paradox_agent_minerals_income]|food=[This.paradox_agent_food]|food_income=[This.paradox_agent_food_income]|consumer_goods=[This.paradox_agent_consumer_goods]|consumer_goods_income=[This.paradox_agent_consumer_goods_income]|alloys=[This.paradox_agent_alloys]|alloys_income=[This.paradox_agent_alloys_income]|unity=[This.paradox_agent_unity]|unity_income=[This.paradox_agent_unity_income]|influence=[This.paradox_agent_influence]"

		every_owned_planet = {
			export_trigger_value_to_variable = { trigger = pop_amount variable = paradox_agent_pops }
			export_trigger_value_to_variable = { trigger = planet_stability variable = paradox_agent_stability }
			export_trigger_value_to_variable = { trigger = free_housing variable = paradox_agent_free_housing }
			export_trigger_value_to_variable = { trigger = free_amenities variable = paradox_agent_free_amenities }
			log = "[PARADOX_AGENT]|PLANET|name=[This.GetName]|pops=[This.paradox_agent_pops]|stability=[This.paradox_agent_stability]|free_housing=[This.paradox_agent_free_housing]|free_amenities=[This.paradox_agent_free_amenities]"
		}

		log = "[PARADOX_AGENT]|STATE_END|date=[GetDate]"
	}
}

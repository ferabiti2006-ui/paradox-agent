namespace = paradox_agent_testbed

# This event first runs after galaxy initialization and then invokes the
# read-only observation effect once per month for the external controller.
country_event = {
	id = paradox_agent_testbed.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
	}

	immediate = {
		if = {
			limit = { NOT = { has_country_flag = paradox_agent_bridge_active } }
			set_country_flag = paradox_agent_bridge_active
			log = "PARADOX_AGENT|BRIDGE_READY|date=[GetDate]|country=[This.GetName]"
		}

		log = "PARADOX_AGENT|STATE_BEGIN|schema=2|date=[GetDate]"
		paradox_agent_export_observation = yes
		log = "PARADOX_AGENT|STATE_END|date=[GetDate]"
	}
}
